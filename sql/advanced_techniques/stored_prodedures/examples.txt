CREATE PROCEDURE GetEmployeeData
    @DepartmentID INT,
    @MinSalary DECIMAL(10,2),
    @Status VARCHAR(20)
AS
BEGIN
    SELECT * FROM Employees 
    WHERE DeptID = @DepartmentID 
      AND Salary >= @MinSalary 
      AND CurrentStatus = @Status;
END;

CREATE PROCEDURE ProcessProductSale
    @ProductID INT,
    @Quantity INT
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @StockAvailable INT;

    -- Look up current stock levels
    SELECT @StockAvailable = StockQuantity FROM Products WHERE ProductID = @ProductID;

    -- Check if we have enough stock using conditional logic
    IF @StockAvailable >= @Quantity
    BEGIN
        BEGIN TRANSACTION;
            -- 1. Deduct the sold units from inventory
            UPDATE Products 
            SET StockQuantity = StockQuantity - @Quantity 
            WHERE ProductID = @ProductID;

            -- 2. Log the transaction details
            INSERT INTO SalesLog (ProductID, QuantitySold, SaleDate)
            VALUES (@ProductID, @Quantity, GETDATE());
        COMMIT TRANSACTION;
        
        PRINT 'Sale successfully processed.';
    END
    ELSE
    BEGIN
        PRINT 'Error: Insufficient inventory to complete the transaction.';
    END
END;
