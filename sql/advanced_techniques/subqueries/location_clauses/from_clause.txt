Find products that have higher that average price of all products

SELECT ProductId, Price, AvgPrice
FROM
(
	SELECT
	ProductID,
	Price,
	AVG(COALESCE(Price, 0)) OVER () AvgPrice
	FROM Sales.Products
)t
WHERE  Price > AvgPrice

Rank customers based on totalamount of sales


SELECT CustomerId, TotalSales,
    RANK() OVER (ORDER BY TotalSales DESC) CustomerRank
FROM
(
	SELECT
	CustomerID,
	SUM(Sales) TotalSales
	FROM Sales.Orders
	GROUP BY CustomerID
)t
ORDER BY TotalSales DESC
