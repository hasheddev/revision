Assigns a rank to each row same values having same rank but leaves no gap in ranks (1, 2, 2, 3)

SELECT
	OrderID,
	ProductID,
	Sales,
	RANK() OVER(ORDER BY Sales DESC) SalesRank,
	DENSE_RANK() OVER(ORDER BY Sales DESC) SalesRank_Dense
FROM Sales.Orders
